\section{Methodology}
% =============================================================================

\begin{frame}{Our Approach: CRISP-DM End-to-End Pipeline}
\centering
\begin{columns}[c]
  \begin{column}{0.18\textwidth}\centering
    \textbf{Data}\\{\small 4 sources\\102K rows\\24 months}
  \end{column}
  \begin{column}{0.02\textwidth}\centering$\rightarrow$\end{column}
  \begin{column}{0.18\textwidth}\centering
    \textbf{Features}\\{\small 24 variables\\RFM + behaviour\\+ email}
  \end{column}
  \begin{column}{0.02\textwidth}\centering$\rightarrow$\end{column}
  \begin{column}{0.18\textwidth}\centering
    \textbf{UMAP}\\{\small 3D reduction\\n\_neighbors=30\\non-linear}
  \end{column}
  \begin{column}{0.02\textwidth}\centering$\rightarrow$\end{column}
  \begin{column}{0.18\textwidth}\centering
    \textbf{GMM K=6}\\{\small full covariance\\10 inits\\seed=42}
  \end{column}
  \begin{column}{0.02\textwidth}\centering$\rightarrow$\end{column}
  \begin{column}{0.18\textwidth}\centering
    \textbf{Strategy}\\{\small CLV model\\ROI scenarios\\roadmap}
  \end{column}
\end{columns}

\vspace{1.0em}
\begin{columns}[c]
  \begin{column}{0.3\textwidth}
    \begin{block}{Why UMAP?}
    \small Preserves non-linear local structure in behavioural data;
    better cluster separation than PCA on this manifold.
    \end{block}
  \end{column}
  \begin{column}{0.3\textwidth}
    \begin{block}{Why GMM?}
    \small Soft probabilistic assignments; handles elliptical, overlapping
    clusters; posterior probability = assignment confidence.
    \end{block}
  \end{column}
  \begin{column}{0.3\textwidth}
    \begin{block}{Why K=6?}
    \small Four convergent criteria: BIC elbow, Silhouette plateau in $K{\in}[5,8]$,
    high ARI stability ($0.84$ vs.\ K=7), 96.5\% posterior $>$0.80;
    operational cap as tie-breaker.
    \end{block}
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{The Active Core: Active Loyalists \& Core Customers}
\begin{columns}[T]
  \begin{column}{0.48\textwidth}
    \textbf{\color{accent}Active Loyalists} --- 30.3\%, n=6{,}495
    \begin{itemize}\small
      \item Recency 131d, freq 5.0/24mo, monetary \euro957
      \item AOV \euro195, broad category footprint (4 cats)
      \item 61.9\% of portfolio dCLV
    \end{itemize}
    \textbf{Strategy:} VIP programme + churn-alert at recency $>$90d.\\
    Investment: \euro45K $\rightarrow$ Uplift: \euro228K (5.1$\times$).
  \end{column}
  \begin{column}{0.48\textwidth}
    \textbf{\color{accent}Core Customers} --- 47.9\%, n=10{,}256
    \begin{itemize}\small
      \item Recency 199d, freq 2.7/24mo $\approx$ 1.36/yr
      \item Monetary \euro484, AOV \euro192
      \item Natural feeder pool for the VIP tier
    \end{itemize}
    \textbf{Strategy:} gamified loyalty with visible tier-progress.\\
    Investment: \euro30K $\rightarrow$ Uplift: \euro95K (3.2$\times$).
  \end{column}
\end{columns}
\vspace{0.3em}
\centering
\includegraphics[width=0.5\textwidth]{assets/clv_analysis.png}
\end{frame}

\begin{frame}{The Engaged \& the Lapsed: Promo-Engaged \& One-Shot Shoppers}
\begin{columns}[T]
  \begin{column}{0.48\textwidth}
    \textbf{\color{accent}Promo-Engaged} --- 3.6\%, n=768
    \begin{itemize}\small
      \item Engagement 0.83 vs.\ portfolio 0.23 (3.7$\times$)
      \item Discount 14\%; freq 1.98/24mo
      \item The brand's most reachable segment
    \end{itemize}
    \textbf{Strategy:} A/B loyalty-points vs.\ discount offers.\\
    Investment: \euro20K $\rightarrow$ Uplift: \euro4K (0.2$\times$).\\[2pt]
    \small\textit{Margin protection, not revenue uplift.}
  \end{column}
  \begin{column}{0.48\textwidth}
    \textbf{\color{accent}One-Shot Shoppers} --- 12.7\%, n=2{,}727
    \begin{itemize}\small
      \item Recency 305d, freq 1.18, n\_categories 1.14
      \item Monetary \euro115, AOV \euro98
      \item Bought once, single-category, then gone
    \end{itemize}
    \textbf{Strategy:} ``Complete the Look'' cross-sell email at 30/60/90d.\\
    Investment: \euro20K $\rightarrow$ Uplift: \euro10K (0.5$\times$).
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{The Dormant Tail: Lapsed Low-Value \& Churned}
\begin{columns}[T]
  \begin{column}{0.48\textwidth}
    \textbf{\color{accent}Lapsed Low-Value} --- 3.2\%, n=675
    \begin{itemize}\small
      \item Recency 315d, freq 1.10
      \item Monetary \euro148, AOV \euro135
      \item Low-cost reactivation candidate
    \end{itemize}
    \textbf{Strategy:} 3-touch win-back sequence (Day 30/60/90).\\
    Investment: \euro10K $\rightarrow$ Uplift: \euro5K (0.5$\times$).
  \end{column}
  \begin{column}{0.48\textwidth}
    \textbf{\color{accent}Churned} --- 2.3\%, n=503
    \begin{itemize}\small
      \item Recency 306d, freq 1.04
      \item Monetary \euro128, engagement 0.14
      \item Smallest cluster; likely lost
    \end{itemize}
    \textbf{Strategy:} 2-touch win-back, then suppress.\\
    Investment: \euro15K $\rightarrow$ Uplift: \euro1K (0.1$\times$).\\[2pt]
    \small\textit{Strategic value: keep the active list clean.}
  \end{column}
\end{columns}
\end{frame}

\section{Conclusions}
% =============================================================================

\begin{frame}{Conclusions \& Limitations}
\begin{columns}[T]
  \begin{column}{0.52\textwidth}
    \textbf{What we found}
    \begin{itemize}
      \item Six behaviourally distinct, interpretable segments confirmed
        by four independent validation criteria (BIC elbow,
        bootstrap Silhouette, ARI stability, posterior confidence)
      \item Active Loyalists alone justify the \euro140K investment at 5.1$\times$ return
      \item Uniform marketing is provably suboptimal: Promo-Engaged
        and Active Loyalists need opposite levers
      \item Portfolio dCLV of \euro16.2M anchors investment sizing
    \end{itemize}
  \end{column}
  \begin{column}{0.45\textwidth}
    \textbf{Honest limitations}
    \begin{itemize}
      \item dCLV is a lower bound: right-censored window
        under-estimates recently acquired customers
      \item ROI figures are scenario estimates, not forecasts;
        all behavioural assumptions require A/B validation
      \item No holdout set: internal validation only
      \item Operational constraint (6 campaigns/cycle) used as
        tie-breaker among $K\in[5,8]$, not primary criterion
    \end{itemize}
  \end{column}
\end{columns}

{\large\color{accent}\textbf{The data says: stop averaging, start differentiating.}}
